model Conversation {
  id        String    @id @default(cuid())
  leadId    String    @unique
  lead      Lead      @relation(fields: [leadId], references: [id])
  channel   String    @default("whatsapp")
  messages  Message[]
  tenantId  String?
  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
}
