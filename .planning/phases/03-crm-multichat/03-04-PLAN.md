<!-- API endpoints from Plan 01 consumed here -->

GET /conversations?limit=50 → Conversation[] (with lead.name, lead.phone, lead.stage, latest message)
GET /conversations/:id/messages → Message[] (role, content, createdAt)
POST /conversations/:id/message → { content, instance } → sends via WhatsApp, returns Message

PATCH /leads/:id/handoff → { override: boolean } → Lead (toggles humanOverride)

GET /pipelines → Pipeline[] with stages[]
POST /pipelines/:id/stages → { name } → Stage
PATCH /pipelines/:pipelineId/stages/:stageId → { name?, order? } → Stage
DELETE /pipelines/:pipelineId/stages/:stageId → void

GET /pipelines/:id/rules → QualificationRule[]
POST /pipelines/:id/rules → { field, operator, value, stageTrigger } → QualificationRule
DELETE /pipelines/:pipelineId/rules/:ruleId → void

```typescript
interface Conversation {
  id: string;
  leadId: string;
  channel: string;
  lead: { name: string | null; phone: string; stage: { name: string } | null; humanOverride: boolean };
  messages: { content: string; role: string; createdAt: string }[];
  updatedAt: string;
}

interface QualificationRule {
  id: string;
  pipelineId: string;
  field: string;    // "creditStatus", "city", "paymentMethod"
  operator: string; // "equals", "contains", "not_equals"
  value: string;
  stageTrigger: string;
}
```
